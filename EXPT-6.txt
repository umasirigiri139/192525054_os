#include <stdio.h>

int main() {
    int n, i, time = 0, completed = 0, highest;
    
    printf("Enter number of processes: ");
    scanf("%d", &n);

    int bt[n], rt[n], pr[n], wt[n], tat[n];

    // Input
    for(i = 0; i < n; i++) {
        printf("Enter Burst Time and Priority for P%d: ", i+1);
        scanf("%d %d", &bt[i], &pr[i]);
        rt[i] = bt[i]; // remaining time
    }

    int finish_time;

    while(completed != n) {
        highest = -1;

        // Find process with highest priority (larger number = higher priority)
        for(i = 0; i < n; i++) {
            if(rt[i] > 0) {
                if(highest == -1 || pr[i] > pr[highest]) {
                    highest = i;
                }
            }
        }

        // Execute for 1 unit time
        rt[highest]--;
        time++;

        // If process completes
        if(rt[highest] == 0) {
            completed++;
            finish_time = time;

            wt[highest] = finish_time - bt[highest];
            if(wt[highest] < 0)
                wt[highest] = 0;
        }
    }

    // Turnaround Time
    for(i = 0; i < n; i++) {
        tat[i] = wt[i] + bt[i];
    }

    // Output
    printf("\nProcess\tBT\tPR\tWT\tTAT\n");
    for(i = 0; i < n; i++) {
        printf("P%d\t%d\t%d\t%d\t%d\n", i+1, bt[i], pr[i], wt[i], tat[i]);
    }

    return 0;
}
