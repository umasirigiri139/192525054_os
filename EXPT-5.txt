#include <stdio.h>

int main() {
    int n, i, j, pos, temp;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    int bt[n], pr[n], wt[n], tat[n];

    // Input burst time and priority
    for(i = 0; i < n; i++) {
        printf("Enter Burst Time and Priority for P%d: ", i+1);
        scanf("%d %d", &bt[i], &pr[i]);
    }

    // Sort based on highest priority (larger number = higher priority)
    for(i = 0; i < n; i++) {
        pos = i;
        for(j = i + 1; j < n; j++) {
            if(pr[j] > pr[pos])
                pos = j;
        }

        // Swap burst time
        temp = bt[i];
        bt[i] = bt[pos];
        bt[pos] = temp;

        // Swap priority
        temp = pr[i];
        pr[i] = pr[pos];
        pr[pos] = temp;
    }

    // Waiting Time
    wt[0] = 0;
    for(i = 1; i < n; i++) {
        wt[i] = wt[i-1] + bt[i-1];
    }

    // Turnaround Time
    for(i = 0; i < n; i++) {
        tat[i] = wt[i] + bt[i];
    }

    // Output
    printf("\nProcess\tBT\tPR\tWT\tTAT\n");
    for(i = 0; i < n; i++) {
        printf("P%d\t%d\t%d\t%d\t%d\n", i+1, bt[i], pr[i], wt[i], tat[i]);
    }

    return 0;
}
