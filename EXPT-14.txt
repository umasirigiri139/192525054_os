#include <stdio.h>
#include <string.h>

#define MAX 10

struct file {
    char name[20];
};

int main() {
    struct file f[MAX];
    int n = 0, choice, i;
    char fname[20];

    while (1) {
        printf("\n1. Create File\n2. Display Files\n3. Search File\n4. Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch (choice) {
            case 1:
                if (n < MAX) {
                    printf("Enter file name: ");
                    scanf("%s", f[n].name);
                    n++;
                    printf("File created successfully\n");
                } else {
                    printf("Directory is full\n");
                }
                break;

            case 2:
                printf("\nFiles in directory:\n");
                for (i = 0; i < n; i++) {
                    printf("%s\n", f[i].name);
                }
                break;

            case 3:
                printf("Enter file name to search: ");
                scanf("%s", fname);
                for (i = 0; i < n; i++) {
                    if (strcmp(fname, f[i].name) == 0) {
                        printf("File found\n");
                        break;
                    }
                }
                if (i == n)
                    printf("File not found\n");
                break;

            case 4:
                return 0;

            default:
                printf("Invalid choice\n");
        }
    }
}
