#include <stdio.h>
#include <stdlib.h>

int main() {
    FILE *fp;

    // Write data (simulate Process 1)
    fp = fopen("shared.txt", "w");
    if (fp == NULL) {
        printf("File error\n");
        return 1;
    }
    fprintf(fp, "Hello from Process 1");
    fclose(fp);

    // Read data (simulate Process 2)
    fp = fopen("shared.txt", "r");
    if (fp == NULL) {
        printf("File error\n");
        return 1;
    }

    char buffer[100];
    fgets(buffer, sizeof(buffer), fp);
    printf("Process 2 read: %s\n", buffer);

    fclose(fp);

    return 0;
}
