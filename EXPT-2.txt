#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>

int main() {
    int fd1, fd2;
    char buffer[100];
    int n;

    // Create and write to source file automatically
    fd1 = open("source.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    write(fd1, "Hello Monisha!", 14);
    close(fd1);

    // Re-open source file for reading
    fd1 = open("source.txt", O_RDONLY);

    // Open destination file
    fd2 = open("dest.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);

    if (fd1 < 0 || fd2 < 0) {
        printf("File error\n");
        return 1;
    }

    // Copy content
    while ((n = read(fd1, buffer, sizeof(buffer))) > 0) {
        write(fd2, buffer, n);
    }

    close(fd1);
    close(fd2);

    printf("File copied successfully\n");

    return 0;
}
