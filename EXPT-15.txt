#include <stdio.h>
#include <string.h>

#define MAX_DIR 5
#define MAX_FILES 5

struct file {
    char fname[20];
};

struct directory {
    char dname[20];
    struct file f[MAX_FILES];
    int fcount;
};

int main() {
    struct directory dir[MAX_DIR];
    int dcount = 0;
    int choice, i, j;
    char dname[20], fname[20];

    while (1) {
        printf("\n1. Create Directory\n2. Create File\n3. Display\n4. Search File\n5. Exit\n");
        printf("Enter choice: ");
        scanf("%d", &choice);

        switch (choice) {

        case 1:
            if (dcount < MAX_DIR) {
                printf("Enter directory name: ");
                scanf("%s", dir[dcount].dname);
                dir[dcount].fcount = 0;
                dcount++;
                printf("Directory created\n");
            } else {
                printf("Directory limit reached\n");
            }
            break;

        case 2:
            printf("Enter directory name: ");
            scanf("%s", dname);

            for (i = 0; i < dcount; i++) {
                if (strcmp(dir[i].dname, dname) == 0) {
                    if (dir[i].fcount < MAX_FILES) {
                        printf("Enter file name: ");
                        scanf("%s", dir[i].f[dir[i].fcount].fname);
                        dir[i].fcount++;
                        printf("File created\n");
                    } else {
                        printf("File limit reached in this directory\n");
                    }
                    break;
                }
            }
            if (i == dcount)
                printf("Directory not found\n");
            break;

        case 3:
            for (i = 0; i < dcount; i++) {
                printf("\nDirectory: %s\n", dir[i].dname);
                for (j = 0; j < dir[i].fcount; j++) {
                    printf("  %s\n", dir[i].f[j].fname);
                }
            }
            break;

        case 4:
            printf("Enter file name to search: ");
            scanf("%s", fname);

            for (i = 0; i < dcount; i++) {
                for (j = 0; j < dir[i].fcount; j++) {
                    if (strcmp(dir[i].f[j].fname, fname) == 0) {
                        printf("File found in directory: %s\n", dir[i].dname);
                        break;
                    }
                }
            }
            break;

        case 5:
            return 0;

        default:
            printf("Invalid choice\n");
        }
    }
}
