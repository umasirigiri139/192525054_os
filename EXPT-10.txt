#include <stdio.h>
#include <string.h>

struct message {
    long type;
    char text[100];
};

int main() {
    struct message msg;

    // Sender
    msg.type = 1;
    strcpy(msg.text, "Hello from Process 1");
    printf("Data sent: %s\n", msg.text);

    // Receiver
    if(msg.type == 1) {
        printf("Data received: %s\n", msg.text);
    }

    return 0;
}
