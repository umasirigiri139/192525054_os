#include <stdio.h>
#include <pthread.h>

// Function for thread
void* thread_func(void* arg) {
    printf("Hello from thread!\n");
    return NULL;
}

int main() {
    pthread_t t1;

    // Create thread
    pthread_create(&t1, NULL, thread_func, NULL);

    // Wait for thread to finish
    pthread_join(t1, NULL);

    printf("Thread execution completed\n");

    return 0;
}
